# =============================================================================
# SuperSonic MultiCustomDictionary 自定义词典样例
# 格式: <word> [<nature> <frequency>]...
# 分隔符: .txt 为空白字符(空格/tab), .csv 为逗号
# nature 编码: 下划线 "_" 分隔业务ID + 元素类型后缀
# 含空格的词用 "#" 占位 (HanlpHelper.SPACE_SPILT = "#")
# 字母词(含大写A-Z)以小写入词典, 大写形式保留为 original
# =============================================================================

# -----------------------------------------------------------------------------
# 1. 数据集名称 (DATASET)
#    nature: _<dataSetId>
#    frequency: 100000 (BaseWordBuilder.DEFAULT_FREQUENCY)
# -----------------------------------------------------------------------------
销售分析 _10 100000
流量看板 _11 100000

# -----------------------------------------------------------------------------
# 2. 指标 (METRIC)
#    nature: _<modelId>_<elementId>_metric
#    反向后缀词: _<modelId>_<elementId>_suffix_metric
#    字母指标(GMV/PV/UV)大写原形会自动保留, 此处正名用小写
# -----------------------------------------------------------------------------
GMV _10_101_metric 100000
销售额 _10_102_metric 100000
PV _10_103_metric 100000
UV _10_104_metric 100000
访问时长 _10_105_metric 100000

# 指标反向词(后缀匹配), word 为正名反转, nature 加 suffix
VMG _10_101_suffix_metric 100000
额售销 _10_102_suffix_metric 100000

# -----------------------------------------------------------------------------
# 3. 维度 (DIMENSION)
#    nature: _<modelId>_<elementId>_dimension
#    反向后缀词: _<modelId>_<elementId>_suffix_dimension
# -----------------------------------------------------------------------------
城市 _10_201_dimension 100000
省份 _10_202_dimension 100000
产品名称 _10_203_dimension 100000
渠道 _10_204_dimension 100000

# 维度反向词
市城 _10_201_suffix_dimension 100000
份省 _10_202_suffix_dimension 100000

# -----------------------------------------------------------------------------
# 4. 维度值 / 枚举值 (VALUE)
#    nature: _<modelId>_<elementId>  (无显式后缀)
#    frequency: 由指标聚合得来 (非固定100000); 白名单值频率另设
#    含空格的值用 # 占位
# -----------------------------------------------------------------------------
北京 _10_201 8000
上海 _10_201 7500
广州 _10_201 6200
深圳 _10_201 5800
广东省 _10_202 9000

# 产品名称维度值
手机 _10_203 12000
笔记本电脑 _10_203 8500
平板电脑 _10_203 4300

# 渠道维度值
官网 _10_204 5000

# -----------------------------------------------------------------------------
# 5. 业务术语 (TERM)
#    nature: _<dataSetId>_<elementId>_term

# -----------------------------------------------------------------------------
环比增长 _10_301_term 100000
